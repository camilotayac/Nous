\definecolor{nouscolor}{HTML}{2980B9}

\AtBeginDocument{%
  \hypersetup{
    linkcolor=nouscolor,
    citecolor=nouscolor,
    urlcolor=nouscolor
  }
}

\newcommand{\nouspagebackground}{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=center, rectangle, fill=nouscolor!20, inner sep=0pt,
          minimum height=\paperheight, minimum width=\paperwidth]
          at (current page.center) {};
  \end{tikzpicture}%
}

\renewcommand{\@makeschapterhead}[1]{%
  \if@tocpage
    \global\@tocpagefalse

    {\raggedright\fontsize{30pt}{30pt}\selectfont\sffamily\bfseries\color{black}#1\par}%

  \else
    \orig@makeschapterhead{#1}%
  \fi
}
